cmake_minimum_required(VERSION 3.16)
project(KVRPC VERSION 2.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
find_package(Threads REQUIRED)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "KVRPC 2 requires Linux epoll/eventfd")
endif()
find_package(Protobuf REQUIRED)
protobuf_generate_cpp(KVRPC_PROTO_SRCS KVRPC_PROTO_HDRS proto/rpc.proto)
option(KVRPC_BUILD_SERVER "Build the Linux KVCache reference server" ON)
option(KVRPC_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(KVRPC_TSAN "Enable ThreadSanitizer" OFF)
if(KVRPC_SANITIZE AND KVRPC_TSAN)
    message(FATAL_ERROR "Select only one sanitizer configuration")
endif()
if(KVRPC_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
elseif(KVRPC_TSAN)
    add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
    add_link_options(-fsanitize=thread)
endif()
add_library(kvrpc src/tcp_connection.cpp src/tcp_server.cpp src/event_loop.cpp src/async_transport.cpp ${KVRPC_PROTO_SRCS})
add_library(KVRPC::kvrpc ALIAS kvrpc)
target_compile_features(kvrpc PUBLIC cxx_std_17)
target_include_directories(kvrpc PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(kvrpc PUBLIC Threads::Threads protobuf::libprotobuf)
if(KVRPC_TSAN)
    include(CheckIncludeFileCXX)
    include(cmake/TSanRuntime.cmake)
endif()
# Export instrumentation flags so installed sanitizer builds link correctly in consumers.
if(KVRPC_SANITIZE)
    target_compile_options(kvrpc INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(kvrpc PUBLIC -fsanitize=address,undefined)
elseif(KVRPC_TSAN)
    target_compile_options(kvrpc INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(kvrpc PUBLIC -fsanitize=thread)
endif()
target_compile_options(kvrpc PRIVATE -Wall -Wextra -Wpedantic)
if(KVRPC_BUILD_SERVER AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_library(kvcache KVCache/src/aof.cpp)
    target_compile_features(kvcache PUBLIC cxx_std_17)
    target_include_directories(kvcache PUBLIC KVCache/include include)
    target_link_libraries(kvcache PUBLIC KVRPC::kvrpc)
    target_compile_options(kvcache PRIVATE -Wall -Wextra -Wpedantic)
    add_executable(kv_server KVCache/src/main.cpp)
    target_link_libraries(kv_server PRIVATE kvcache)
    install(TARGETS kv_server RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
foreach(name rpc_server rpc_client)
    add_executable(kvrpc_${name} examples/${name}.cpp)
    target_link_libraries(kvrpc_${name} PRIVATE KVRPC::kvrpc)
endforeach()
add_executable(profile_demo examples/profile_demo.cpp)
target_link_libraries(profile_demo PRIVATE KVRPC::kvrpc)
add_executable(kv_bench benchmarks/kv_bench.cpp)
target_link_libraries(kv_bench PRIVATE KVRPC::kvrpc)
add_executable(kvrpc_example examples/kvcache.cpp)
target_link_libraries(kvrpc_example PRIVATE KVRPC::kvrpc)
if(BUILD_TESTING)
    if(KVRPC_TSAN)
        find_package(Python3 COMPONENTS Interpreter REQUIRED)
        add_executable(test_tsan_exception_runtime tests/diagnostics/future_exception_tsan.cpp)
        target_link_libraries(test_tsan_exception_runtime PRIVATE KVRPC::kvrpc)
        add_test(NAME tsan_exception_runtime COMMAND test_tsan_exception_runtime)
        set_tests_properties(tsan_exception_runtime PROPERTIES TIMEOUT 30)
        add_executable(test_tsan_race tests/diagnostics/intentional_race.cpp)
        target_link_libraries(test_tsan_race PRIVATE KVRPC::kvrpc)
        add_test(NAME tsan_detector COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/diagnostics/check_tsan_detector.py $<TARGET_FILE:test_tsan_race>)
        set_tests_properties(tsan_detector PROPERTIES TIMEOUT 30)
    endif()
    foreach(name serializer connection_pool rpc_client rpc_server kvcache_client transport executor reactor)
        add_executable(test_${name} tests/test_${name}.cpp)
        target_link_libraries(test_${name} PRIVATE KVRPC::kvrpc)
        target_compile_options(test_${name} PRIVATE -Wall -Wextra -Wpedantic)
        add_test(NAME ${name} COMMAND test_${name})
        set_tests_properties(${name} PROPERTIES TIMEOUT 30)
    endforeach()
    if(TARGET kv_server)
        add_executable(test_storage_v2 tests/test_storage_v2.cpp)
        target_link_libraries(test_storage_v2 PRIVATE kvcache)
        add_test(NAME storage_v2 COMMAND test_storage_v2)
        set_tests_properties(storage_v2 PROPERTIES TIMEOUT 30)
        add_executable(test_storage tests/test_storage.cpp)
        target_link_libraries(test_storage PRIVATE kvcache)
        add_test(NAME storage COMMAND test_storage)
        set_tests_properties(storage PROPERTIES TIMEOUT 30)
        find_package(Python3 COMPONENTS Interpreter REQUIRED)
        add_test(NAME server_integration COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_server.py $<TARGET_FILE:kv_server> $<TARGET_FILE:kvrpc_example> group $<TARGET_FILE:profile_demo> ${CMAKE_CURRENT_SOURCE_DIR}/examples/profiles.tsv)
        add_test(NAME server_always COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_server.py $<TARGET_FILE:kv_server> $<TARGET_FILE:kvrpc_example> always $<TARGET_FILE:profile_demo> ${CMAKE_CURRENT_SOURCE_DIR}/examples/profiles.tsv)
        set_tests_properties(server_integration server_always PROPERTIES TIMEOUT 45)
    endif()
endif()
install(TARGETS kvrpc EXPORT KVRPCTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT KVRPCTargets NAMESPACE KVRPC:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/KVRPC)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/KVRPCConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/KVRPCConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/KVRPC)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/KVRPCConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/KVRPCConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/KVRPCConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/KVRPC)

install(FILES ${KVRPC_PROTO_HDRS} DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES proto/rpc.proto DESTINATION ${CMAKE_INSTALL_DATADIR}/kvrpc/proto)
