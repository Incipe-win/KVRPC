cmake_minimum_required(VERSION 3.16)
project(KVRPC VERSION 1.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
find_package(Threads REQUIRED)
option(KVRPC_BUILD_SERVER "Build the Linux KVCache reference server" ON)
option(KVRPC_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(KVRPC_TSAN "Enable ThreadSanitizer" OFF)
if(KVRPC_SANITIZE AND KVRPC_TSAN)
    message(FATAL_ERROR "Select only one sanitizer configuration")
endif()
if(KVRPC_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
elseif(KVRPC_TSAN)
    add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
    add_link_options(-fsanitize=thread)
endif()
add_library(kvrpc src/tcp_connection.cpp)
add_library(KVRPC::kvrpc ALIAS kvrpc)
target_compile_features(kvrpc PUBLIC cxx_std_17)
target_include_directories(kvrpc PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(kvrpc PUBLIC Threads::Threads)
# Export instrumentation flags so installed sanitizer builds link correctly in consumers.
if(KVRPC_SANITIZE)
    target_compile_options(kvrpc INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(kvrpc PUBLIC -fsanitize=address,undefined)
elseif(KVRPC_TSAN)
    target_compile_options(kvrpc INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(kvrpc PUBLIC -fsanitize=thread)
endif()
target_compile_options(kvrpc PRIVATE -Wall -Wextra -Wpedantic)
if(KVRPC_BUILD_SERVER AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_library(kvcache KVCache/src/tcp_server.cpp KVCache/src/aof.cpp)
    target_compile_features(kvcache PUBLIC cxx_std_17)
    target_include_directories(kvcache PUBLIC KVCache/include include)
    target_link_libraries(kvcache PUBLIC Threads::Threads)
    target_compile_options(kvcache PRIVATE -Wall -Wextra -Wpedantic)
    add_executable(kv_server KVCache/src/main.cpp)
    target_link_libraries(kv_server PRIVATE kvcache)
    install(TARGETS kv_server RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
add_executable(kvrpc_example examples/kvcache.cpp)
target_link_libraries(kvrpc_example PRIVATE KVRPC::kvrpc)
if(BUILD_TESTING)
    foreach(name serializer connection_pool rpc_client kvcache_client transport executor)
        add_executable(test_${name} tests/test_${name}.cpp)
        target_link_libraries(test_${name} PRIVATE KVRPC::kvrpc)
        target_compile_options(test_${name} PRIVATE -Wall -Wextra -Wpedantic)
        add_test(NAME ${name} COMMAND test_${name})
        set_tests_properties(${name} PROPERTIES TIMEOUT 30)
    endforeach()
    if(TARGET kv_server)
        add_executable(test_storage tests/test_storage.cpp)
        target_link_libraries(test_storage PRIVATE kvcache)
        add_test(NAME storage COMMAND test_storage)
        set_tests_properties(storage PROPERTIES TIMEOUT 30)
        find_package(Python3 COMPONENTS Interpreter REQUIRED)
        add_test(NAME server_integration COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_server.py $<TARGET_FILE:kv_server> $<TARGET_FILE:kvrpc_example>)
        set_tests_properties(server_integration PROPERTIES TIMEOUT 45)
    endif()
endif()
install(TARGETS kvrpc EXPORT KVRPCTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT KVRPCTargets NAMESPACE KVRPC:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/KVRPC)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/KVRPCConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/KVRPCConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/KVRPC)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/KVRPCConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/KVRPCConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/KVRPCConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/KVRPC)
